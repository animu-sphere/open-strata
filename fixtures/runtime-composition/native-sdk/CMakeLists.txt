cmake_minimum_required(VERSION 3.20)
project(TinySdk VERSION 1.0.0 LANGUAGES CXX)
add_library(tiny SHARED tiny.cpp)
target_include_directories(tiny PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>" "$<INSTALL_INTERFACE:include>")
set_target_properties(tiny PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(sdk-inspect main.cpp)
target_link_libraries(sdk-inspect PRIVATE tiny)
install(TARGETS tiny sdk-inspect EXPORT TinyTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES tiny.h DESTINATION include)
install(EXPORT TinyTargets NAMESPACE Tiny:: DESTINATION lib/cmake/Tiny)
install(FILES TinyConfig.cmake DESTINATION lib/cmake/Tiny)
install(FILES plugInfo.json generatedSchema.usda DESTINATION plugins/Tiny)
